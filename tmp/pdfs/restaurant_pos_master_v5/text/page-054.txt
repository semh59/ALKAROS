identity.permissions: permission_id PK, code unique not null,
description not null.
Examples: Order.Create, Order.Void, Payment.Approve, Bill.Split,
Inventory.Adjust, Invoice.Create, Cash.Close.

identity.user_roles: user_role_id PK, user_id FK, role_id FK, created_at.
Constraint: unique(user_id, role_id).

identity.role_permissions: role_permission_id PK, role_id FK,
