III.12.3 recipe_ingredients
recipe.recipe_ingredients: recipe_ingredient_id PK, recipe_version_id FK
not null, ingredient_product_id FK not null, quantity not null, unit_code not
null, waste_factor default 0, mandatory default true.

recipe.unit_conversions: unit_conversion_id PK, from_unit_code not null,

null, created_at not null.
